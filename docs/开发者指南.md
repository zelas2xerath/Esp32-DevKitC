```cpp
class SensorBase {
```

```cpp
// lib/SensorManager/NewSensor.cpp
#include "NewSensor.h"
#include <LogManager/LogManager.h>

NewSensor::NewSensor(int pin) 
    : _pin(pin), _readCount(0), _lastErrorCode(ErrorCode::SUCCESS) {
    LOG_TRACE("NewSensor构造函数: 引脚=%d", pin);
}

NewSensor::~NewSensor() {
    LOG_TRACE("NewSensor析构函数");
}

bool NewSensor::begin() {
    LOG_TRACE("初始化NewSensor，引脚: %d", _pin);
    
    if (_pin < 0 || _pin > 39) {
        setError(ErrorCode::INVALID_PARAMETER, "无效的引脚号: " + String(_pin));
        return false;
    }
    
    if (!initializeHardware()) {
        setError(ErrorCode::SENSOR_INIT_FAILED, "硬件初始化失败");
        return false;
    }
    
    _lastErrorCode = ErrorCode::SUCCESS;
    _lastErrorMessage = "";
    
    LOG_NOTICE("NewSensor初始化成功");
    return true;
}

int NewSensor::readSensorData() {
    if (_lastErrorCode != ErrorCode::SUCCESS) {
        LOG_WARNING("传感器未正确初始化，跳过读取");
        return -1;
    }
    
    int rawValue = readRawData();
    if (rawValue < 0) {
        setError(ErrorCode::SENSOR_READ_FAILED, "读取原始数据失败");
        return -1;
    }
    
    if (!isDataValid(rawValue)) {
        setError(ErrorCode::SENSOR_OUT_OF_RANGE, "数据超出有效范围: " + String(rawValue));
        return -1;
    }
    
    _readCount++;
    _lastErrorCode = ErrorCode::SUCCESS;
    _lastErrorMessage = "";
    
    LOG_VERBOSE("NewSensor读取成功: %d (第%lu次)", rawValue, _readCount);
    return rawValue;
}

bool NewSensor::initializeHardware() {
    // 实现具体的硬件初始化逻辑
    pinMode(_pin, INPUT);
    delay(100);  // 等待硬件稳定
    return true;
}

int NewSensor::readRawData() {
    // 实现具体的数据读取逻辑
    return analogRead(_pin);
}

bool NewSensor::isDataValid(int value) {
    // 实现数据验证逻辑
    return (value >= 0 && value <= 4095);
}

ErrorCode NewSensor::getLastErrorCode() const {
    return _lastErrorCode;
}

String NewSensor::getLastErrorMessage() const {
    return _lastErrorMessage;
}

unsigned long NewSensor::getReadCount() const {
    return _readCount;
}
```

```cpp
void readNewSensor() {
    if (!rtosSystemManager->getNewSensor()) {
        return;
    }
    
    NewSensor* newSensor = rtosSystemManager->getNewSensor();
    int sensorValue = newSensor->readSensorData();
    
    if (sensorValue >= 0 && NewSensor::isDataValid(sensorValue)) {
        // 创建数据结构
        NewSensorData data;
        data.value = sensorValue;
        data.readCount = newSensor->getReadCount();
        
        // 发送到队列
        SensorDataMessage message = createSensorMessage(SENSOR_MSG_NEW_SENSOR, &data);
        if (!safeSendToQueue(xSensorDataQueue, &message, pdMS_TO_TICKS(QUEUE_SEND_TIMEOUT_MS))) {
            LOG_WARNING("发送NewSensor数据到队列失败");
        } else {
            LOG_VERBOSE("NewSensor数据: %d", sensorValue);
        }
    } else {
        // 错误处理
        ErrorData errorData;
        errorData.errorCode = newSensor->getLastErrorCode();
        strncpy(errorData.errorMsg, newSensor->getLastErrorMessage().c_str(), 63);

        safeSendToQueue(xSensorDataQueue, &errorMessage, pdMS_TO_TICKS(QUEUE_SEND_TIMEOUT_MS));
    }
}
```

## 🔍 自定义滤波算法

### 1. 创建自定义滤波器

#### 定义滤波器接口
```cpp
// lib/DataProcessing/CustomFilter.h
#pragma once

#include <vector>
#include "DataFilterConfig.h"

template<typename T>
class CustomFilter {
private:
    std::vector<T> _dataBuffer;
    size_t _windowSize;
    size_t _currentIndex;
    bool _bufferFull;
    
public:
    CustomFilter(size_t windowSize) 
        : _windowSize(windowSize), _currentIndex(0), _bufferFull(false) {
        _dataBuffer.resize(windowSize);
    }
    
    T filter(T newValue) {
        // 添加新数据到缓冲区
        _dataBuffer[_currentIndex] = newValue;
        _currentIndex = (_currentIndex + 1) % _windowSize;
        
        if (!_bufferFull && _currentIndex == 0) {
            _bufferFull = true;
        }
        
        // 实现自定义滤波算法
        return applyCustomAlgorithm();
    }
    
    void reset() {
        _currentIndex = 0;
        _bufferFull = false;
        std::fill(_dataBuffer.begin(), _dataBuffer.end(), T{});
    }
    
private:
    T applyCustomAlgorithm() {
        if (!_bufferFull && _currentIndex < 2) {
            return _dataBuffer[_currentIndex > 0 ? _currentIndex - 1 : 0];
        }
        
        // 示例：加权移动平均算法
        T sum = T{};
        T weightSum = T{};
        size_t dataCount = _bufferFull ? _windowSize : _currentIndex;
        
        for (size_t i = 0; i < dataCount; i++) {
            T weight = static_cast<T>(i + 1);  // 线性权重
            sum += _dataBuffer[i] * weight;
            weightSum += weight;
        }
        
        return (weightSum > 0) ? (sum / weightSum) : T{};
    }
};
```

#### 内存监控工具
```cpp
// lib/DebugTools/MemoryMonitor.h
class MemoryMonitor {
public:
    static void printMemoryUsage(const String& location) {
        Serial.printf("[MEM] %s - 自由堆: %d bytes, 最小自由堆: %d bytes\n",
                      location.c_str(),
                      ESP.getFreeHeap(),
                      ESP.getMinFreeHeap());
    }
    
    static void checkMemoryLeak() {
        static size_t lastFreeHeap = ESP.getFreeHeap();
        size_t currentFreeHeap = ESP.getFreeHeap();
        
        if (currentFreeHeap < lastFreeHeap) {
            Serial.printf("[MEM] 可能的内存泄漏: %d bytes\n", 
                          lastFreeHeap - currentFreeHeap);
        }
        
        lastFreeHeap = currentFreeHeap;
    }
};
```
