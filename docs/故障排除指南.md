   ```cpp
       }
   }
   ```

2. **连接状态监控**
   ```cpp
   void monitorWiFiStatus() {
       wl_status_t status = WiFi.status();
       switch (status) {
           case WL_CONNECTED:
               Serial.println("WiFi已连接");
               break;
           case WL_NO_SSID_AVAIL:
               Serial.println("找不到指定的SSID");
               break;
           case WL_CONNECT_FAILED:
               Serial.println("WiFi连接失败");
               break;
           case WL_CONNECTION_LOST:
               Serial.println("WiFi连接丢失");
               break;
           default:
               Serial.printf("WiFi状态: %d\n", status);
       }
   }
   ```

#### 解决方案
```cpp
// 1. 改进WiFi连接逻辑
bool connectToWiFi(const char* ssid, const char* password, int timeout = 30000) {
    WiFi.begin(ssid, password);
    
    unsigned long startTime = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - startTime < timeout) {
        delay(500);
        Serial.print(".");
    }
    
    if (WiFi.status() == WL_CONNECTED) {
        Serial.printf("\nWiFi连接成功! IP: %s\n", WiFi.localIP().toString().c_str());
        return true;
    } else {
        Serial.println("\nWiFi连接超时");
        return false;
    }
}

// 2. 添加WiFi重连机制
void handleWiFiReconnect() {
    static unsigned long lastReconnectAttempt = 0;
    static int reconnectAttempts = 0;
    
    if (WiFi.status() != WL_CONNECTED) {
        unsigned long currentTime = millis();
        if (currentTime - lastReconnectAttempt >= 30000) { // 30秒重试一次
            lastReconnectAttempt = currentTime;
            reconnectAttempts++;
            
            Serial.printf("WiFi重连尝试 #%d\n", reconnectAttempts);
            WiFi.reconnect();
            
            if (reconnectAttempts >= 5) {
                // 5次重连失败后重启设备
                Serial.println("WiFi重连失败，重启设备");
                ESP.restart();
            }
        }
    } else {
        reconnectAttempts = 0; // 重置重连计数
    }
}
```

```cpp
        attempts++;
        Serial.printf("MQTT连接尝试 #%d\n", attempts);
```

```cpp
// 1. 传感器重新初始化

    bool allSuccess = true;

    delay(100);
    Wire.begin(21, 19); // SDA, SCL
```

### 模拟传感器问题

#### 症状

#### 解决方案
```cpp
// 1. ADC多次采样平均
float readAnalogSensorAverage(int pin, int samples = 10) {
    long sum = 0;
    for (int i = 0; i < samples; i++) {
        sum += analogRead(pin);
        delay(10);
    }
    return (float)sum / samples;
}

// 2. 数字滤波
class DigitalFilter {
private:
    float alpha;
    float filteredValue;
    bool initialized;
    
public:
    DigitalFilter(float smoothingFactor = 0.1) : alpha(smoothingFactor), initialized(false) {}
    
    float filter(float newValue) {
        if (!initialized) {
            filteredValue = newValue;
            initialized = true;
        } else {
            filteredValue = alpha * newValue + (1 - alpha) * filteredValue;
        }
        return filteredValue;
    }
};

// 3. 异常值检测
class OutlierDetector {
private:
    float lastValue;
    float threshold;
    
public:
    OutlierDetector(float detectionThreshold = 0.2) : threshold(detectionThreshold) {}
    
    bool isOutlier(float newValue) {
        if (abs(newValue - lastValue) / lastValue > threshold) {
            return true;
        }
        lastValue = newValue;
        return false;
    }
};
```

```cpp
class ConfigBackup {
private:
    Preferences prefs;
    
public:
    void backupConfig() {
        prefs.begin("backup", false);
        
        // 备份WiFi配置
        prefs.putString("wifi_ssid", WiFi.SSID());
        prefs.putString("wifi_pass", getWiFiPassword());
        
        // 备份MQTT配置

        prefs.putInt("mqtt_port", mqttPort);
        
        prefs.end();

            String pass = prefs.getString("wifi_pass");
            
            // 恢复WiFi配置
            WiFi.begin(ssid.c_str(), pass.c_str());
            
            prefs.end();
            return true;
        }
        
        prefs.end();
        return false;
    }
};
```

#### 症状

```cpp
// 1. 内存监控
void monitorMemoryUsage() {
    size_t freeHeap = ESP.getFreeHeap();
    size_t minFreeHeap = ESP.getMinFreeHeap();
    size_t heapSize = ESP.getHeapSize();
    
    Serial.printf("堆内存状态:\n");
    Serial.printf("  总大小: %d 字节\n", heapSize);
    Serial.printf("  当前空闲: %d 字节\n", freeHeap);
    Serial.printf("  最小空闲: %d 字节\n", minFreeHeap);
    Serial.printf("  使用率: %.1f%%\n", (float)(heapSize - freeHeap) / heapSize * 100);
    
    if (freeHeap < 10000) {
        Serial.println("警告: 内存不足!");
        // 触发内存清理
        performMemoryCleanup();
    }
}

// 2. 内存清理
void performMemoryCleanup() {
    // 清理消息缓冲区
    messageBuffer.clear();
    
    // 强制垃圾回收
    heap_caps_check_integrity_all(true);
    
    // 重新分配关键内存
    reallocateCriticalBuffers();
}

// 3. 内存池管理
class MemoryPool {
private:
    void* pool;
    size_t poolSize;
    size_t used;
    
public:
    MemoryPool(size_t size) : poolSize(size), used(0) {
        pool = malloc(size);
    }
    
    void* allocate(size_t size) {
        if (used + size <= poolSize) {
            void* ptr = (char*)pool + used;
            used += size;
            return ptr;
        }
        return nullptr;
    }
    
    void reset() {
        used = 0;
    }
    
    ~MemoryPool() {
        free(pool);
    }
};
```

## 📊 系统监控和诊断

### 实时监控工具

```cpp
class SystemMonitor {
private:
    unsigned long lastReport = 0;
    const unsigned long reportInterval = 30000; // 30秒
    
public:
    void update() {
        unsigned long currentTime = millis();
        if (currentTime - lastReport >= reportInterval) {
            generateSystemReport();
            lastReport = currentTime;
        }
    }
    
    void generateSystemReport() {
        JsonDocument report;
        
        // 系统信息
        report["uptime"] = millis() / 1000;
        report["free_heap"] = ESP.getFreeHeap();
        report["min_free_heap"] = ESP.getMinFreeHeap();
        report["cpu_freq"] = getCpuFrequencyMhz();
        
        // 网络信息
        report["wifi_rssi"] = WiFi.RSSI();
        report["wifi_connected"] = WiFi.isConnected();
        report["mqtt_connected"] = mqttClient.connected();
        
        // 任务信息
        JsonObject tasks = report.createNestedObject("tasks");
        tasks["sensor"] = (xSensorTaskHandle != nullptr);
        tasks["network"] = (xNetworkTaskHandle != nullptr);
        tasks["control"] = (xControlTaskHandle != nullptr);
        
        // 发送报告
        String reportStr;
        serializeJson(report, reportStr);
        Serial.println("系统报告: " + reportStr);
        
        // 通过MQTT发送
        if (mqttClient.connected()) {
            mqttClient.publish("agriculture/ESP32_001/status/system", reportStr.c_str());
        }
    }
};
```
